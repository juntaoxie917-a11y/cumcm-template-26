% ===============常用数字集合与字母===============

% 黑板粗体（数集）
\newcommand{\RR}{\mathbb{R}}
\newcommand{\QQ}{\mathbb{Q}}
\newcommand{\ZZ}{\mathbb{Z}}
\newcommand{\CC}{\mathbb{C}}
\newcommand{\NN}{\mathbb{N}}

% 花体字母（集合、空间等）
\newcommand{\cA}{\mathcal{A}}
\newcommand{\cB}{\mathcal{B}}
\newcommand{\cF}{\mathcal{F}}
\newcommand{\cL}{\mathcal{L}}

% ===============自动伸缩的定界符===============

\newcommand{\pr}[1]{\left(#1\right)}          % 圆括号
\newcommand{\br}[1]{\left[#1\right]}          % 方括号
\renewcommand{\brace}[1]{\left\{#1\right\}}     % 大括号
\newcommand{\abs}[1]{\left\lvert #1 \right\rvert}  % 绝对值
\newcommand{\norm}[1]{\left\lVert #1 \right\rVert} % 范数
\newcommand{\floor}[1]{\left\lfloor #1 \right\rfloor}  % 向下取整
\newcommand{\ceil}[1]{\left\lceil #1 \right\rceil}     % 向上取整

% ===============微积分与求导===============

\newcommand{\dd}[2]{\frac{\mathrm{d} #1}{\mathrm{d} #2}}       % 全导数
\newcommand{\pd}[2]{\frac{\partial #1}{\partial #2}}            % 偏导数
\newcommand{\pdd}[2]{\frac{\partial^2 #1}{\partial #2^2}}       % 二阶偏导
\newcommand{\inv}[1]{{#1}^{-1}}                                  % 逆

% ===============常用文本指令===============

% 文本指令仅在数学模式下使用
\newcommand{\st}{\text{s.t.}}              % subject to
\newcommand{\ie}{\text{i.e., }}            % 即
\newcommand{\eg}{\text{e.g., }}            % 例如
\newcommand{\etc}{\text{etc.}}             % 等等
\newcommand{\wrt}{\text{w.r.t. }}          % with respect to

% ===============概率与统计===============

\newcommand{\E}[1]{\mathbb{E}\!\left[#1\right]}                 % 期望
\newcommand{\Var}[1]{\operatorname{Var}\!\left(#1\right)}       % 方差
\newcommand{\Cov}[2]{\operatorname{Cov}\!\left(#1,#2\right)}    % 协方差
\newcommand{\given}{\;\middle|\;}                                % 条件竖线

% ===============数学运算符===============

\DeclareMathOperator*{\argmax}{arg\,max}
\DeclareMathOperator*{\argmin}{arg\,min}
\DeclareMathOperator{\tr}{tr}              % 迹
\DeclareMathOperator{\rank}{rank}          % 秩
\DeclareMathOperator{\diag}{diag}          % 对角

% ===============加粗向量与矩阵===============

\newcommand{\vb}[1]{\mathbf{#1}}           % 通用加粗向量
\newcommand{\mtx}[1]{\mathbf{#1}}          % 矩阵（可与向量区分风格）
